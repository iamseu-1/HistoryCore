use serde::{Deserialize, Serialize};
use unicode_normalization::UnicodeNormalization;

/// 서재가 아는 이름공간 접두어(`분류:조선`의 `분류`). 서재 밖 자료(연표 · 字 · 책)도 든다.
/// `history_library::RESERVED_PREFIXES`와 같아야 한다.
pub const NAMESPACES: [&str; 8] = ["문서", "분류", "틀", "파일", "대문", "연표", "字", "책"];
pub const DEFAULT_NS: &str = "문서";
pub const CATEGORY: &str = "분류";
pub const TEMPLATE: &str = "틀";
pub const FILE: &str = "파일";
pub const TIMELINE: &str = "연표";

/// 이름공간과 제목(+앵커)으로 가리키는 대상. 같은 문서 안의 문단을 가리키면 `ns` · `key`가 비어 있다.
#[derive(Clone, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub struct ObjectRef {
    pub ns: String,
    pub key: String,
    pub anchor: Option<String>,
}

impl ObjectRef {
    pub fn new(ns: &str, key: &str) -> ObjectRef {
        ObjectRef { ns: ns.to_string(), key: key.to_string(), anchor: None }
    }

    /// 화면에 보이는 제목 — 일반 문서는 접두어를 붙이지 않는다.
    pub fn title(&self) -> String {
        if self.ns == DEFAULT_NS {
            self.key.clone()
        } else {
            format!("{}:{}", self.ns, self.key)
        }
    }

    /// 앵커를 뺀 비교용 키 `이름공간:제목`.
    pub fn path(&self) -> String {
        format!("{}:{}", self.ns, self.key)
    }

    pub fn without_anchor(&self) -> ObjectRef {
        ObjectRef { ns: self.ns.clone(), key: self.key.clone(), anchor: None }
    }

    pub fn is_same_page(&self) -> bool {
        self.key.is_empty()
    }

    /// `틀:조선의 전쟁#인자` 같은 글을 주소로 — 옛 Dart `ObjectRef.tryParse`와 같다. NFC로 맞춘다.
    /// 콜론 앞이 [NAMESPACES]에 있을 때만 이름공간으로 가른다. 쓸 글자가 없으면 None.
    pub fn parse(text: &str) -> Option<ObjectRef> {
        let nfc: String = text.trim().nfc().collect();
        let mut raw = nfc.as_str();
        if let Some(rest) = raw.strip_prefix(':') {
            raw = rest.trim();
        }
        if raw.is_empty() {
            return None;
        }
        let mut anchor = None;
        if let Some(hash) = raw.find('#') {
            let a = raw[hash + 1..].trim();
            if !a.is_empty() {
                anchor = Some(a.to_string());
            }
            raw = raw[..hash].trim();
        }
        if raw.is_empty() {
            return None;
        }
        let mut ns = DEFAULT_NS;
        if let Some(colon) = raw.find(':') {
            let head = raw[..colon].trim();
            if colon > 0 && NAMESPACES.contains(&head) {
                ns = NAMESPACES.iter().find(|n| **n == head).copied().unwrap_or(DEFAULT_NS);
                raw = raw[colon + 1..].trim();
            }
        }
        if raw.is_empty() {
            return None;
        }
        Some(ObjectRef { ns: ns.to_string(), key: raw.to_string(), anchor })
    }
}

use crate::ast::Inline;

pub fn plain_inline(nodes: &[Inline]) -> String {
    let mut out = String::new();
    walk_inline(nodes, &mut out);
    out
}

fn walk_inline(nodes: &[Inline], out: &mut String) {
    for n in nodes {
        match n {
            Inline::Text(t) | Inline::Code(t) => out.push_str(t),
            Inline::Styled { children, .. } | Inline::Size { children, .. } => walk_inline(children, out),
            Inline::Link { target, label } => match label {
                Some(l) => walk_inline(l, out),
                None if target.key.is_empty() => out.push_str(target.anchor.as_deref().unwrap_or("")),
                None => out.push_str(&target.title()),
            },
            Inline::Url { uri, label } => match label {
                Some(l) => walk_inline(l, out),
                None => out.push_str(uri),
            },
            Inline::Ruby { base, ruby } => {
                out.push_str(base);
                out.push('(');
                out.push_str(ruby);
                out.push(')');
            }
            Inline::Break => out.push(' '),
            Inline::Param { name, fallback } => out.push_str(fallback.as_deref().unwrap_or(name)),
            Inline::File { .. } | Inline::FootnoteRef { .. } | Inline::Macro { .. } => {}
        }
    }
}

    ("^^", Style::Superscript),
    (",,", Style::Subscript),
];

#[derive(Clone, Debug)]
pub(crate) struct Line {
    pub(crate) text: String,
    pub(crate) offset: usize,
}

struct HeadingInfo {
    level: u8,
    offset: usize,
    title: String,
}

pub(crate) struct Entry {
    indent: usize,
    marker: String,
    start_at: Option<u32>,
    text: String,
}

pub(crate) struct Run {
    pub(crate) opt: Options,
    source: String,
    footnotes: Vec<FootnoteDef>,
    named: HashMap<String, u32>,
    categories: Vec<String>,
    headings: Vec<HeadingInfo>,
    section_counter: usize,
}

/// 문단 제목 `== 제목 ==` · 접힌 `==# 제목 #==`. Dart 정규식 `^(={1,6})(#?)\s*(.+?)\s*\2\1\s*$`를
/// 역참조 없이 — 등호 수를 많은 쪽부터, `#`은 있는 쪽부터 맞춰 본다(정규식의 되돌아가기와 같은 차례).
pub(crate) fn heading(line: &str) -> Option<(u8, bool, &str)> {
    let lead = line.bytes().take_while(|&b| b == b'=').count();
    for n in (1..=lead.min(6)).rev() {
        let after = &line[n..];
        for folded in [true, false] {
            let body = if folded {
                match after.strip_prefix('#') {
                    Some(b) => b,
                    None => continue,
                }
            } else {
                after
            };
            let close = format!("{}{}", if folded { "#" } else { "" }, "=".repeat(n));
            let end = body.trim_end();
            if let Some(inner) = end.strip_suffix(close.as_str()) {
                let title = inner.trim();
                if !title.is_empty() {
                    return Some((n as u8, folded, title));
                }
            }
        }
    }
    None
}

pub(crate) fn is_table_start(t: &str) -> bool {
    t.starts_with("||") || (CAPTION.is_match(t) && t.trim_end().ends_with("||"))
}

fn split_lines(source: &str) -> Vec<Line> {
    let mut lines = Vec::new();
    let mut offset = 0;
    for text in source.split('\n') {
        lines.push(Line { text: text.to_string(), offset });
        offset += text.len() + 1;
    }
    lines
}

/// 앞쪽 공백을 많아야 `n`바이트 걷어 낸다 — 문자 경계를 넘지 않는다(전각 공백 등).
fn dedent(t: &str, n: usize) -> (&str, usize) {
    let mut cut = 0;
    for (i, c) in t.char_indices() {
        if i >= n || !c.is_whitespace() || i + c.len_utf8() > n {
            break;
        }
        cut = i + c.len_utf8();
    }
    (&t[cut..], cut)
}

fn has_content(inlines: &[Inline]) -> bool {
    inlines.iter().any(|n| match n {
        Inline::Text(t) => !t.trim().is_empty(),
        Inline::Break => false,
        _ => true,
    })
}

fn flush(buf: &mut String, out: &mut Vec<Inline>) {
    if !buf.is_empty() {
        out.push(Inline::Text(std::mem::take(buf)));
    }
}

/// 쉼표로 나누되 `\` 이스케이프와 `[ ]` · `( )` 안의 쉼표는 건드리지 않는다.
pub(crate) fn split_args(s: &str) -> Vec<String> {
    let mut parts = Vec::new();
    let mut buf = String::new();
    let mut bracket = 0usize;
    let mut chars = s.chars().peekable();
    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(next) = chars.next() {
                buf.push(next);
                continue;
            }
        }
        if c == '[' || c == '(' {
            bracket += 1;
        }
        if (c == ']' || c == ')') && bracket > 0 {
            bracket -= 1;
        }
        if c == ',' && bracket == 0 {
            parts.push(buf.trim().to_string());
            buf.clear();
            continue;
        }
        buf.push(c);
    }
    if !buf.is_empty() || !parts.is_empty() {
        parts.push(buf.trim().to_string());
    }
    parts
}

pub(crate) fn key_values(args: &[String]) -> BTreeMap<String, String> {
    let mut out = BTreeMap::new();
    for a in args {
        if let Some(eq) = a.find('=') {
            if eq > 0 {
                out.insert(a[..eq].trim().to_string(), a[eq + 1..].trim().to_string());
            }
        }
    }
    out
}

fn options(label: Option<&str>) -> BTreeMap<String, String> {
    let mut out = BTreeMap::new();
    if let Some(label) = label {
        for part in label.split('&') {
            if let Some(eq) = part.find('=') {
                if eq > 0 {
                    out.insert(part[..eq].trim().to_string(), part[eq + 1..].trim().to_string());
                }
            }
        }
    }
    out
}

/// `sep`으로 나누되 `[[ ]]` · `{{{ }}}` · `[ ]` 안과 `\` 뒤는 건드리지 않는다.
pub(crate) fn split_top_level(s: &str, sep: &str) -> Vec<String> {
    let b = s.as_bytes();
    let sep_b = sep.as_bytes();
    let (mut link, mut brace, mut bracket) = (0usize, 0usize, 0usize);
    let mut out = Vec::new();
    let (mut last, mut i) = (0usize, 0usize);
    while i < b.len() {
        if b[i] == b'\\' {
            i += 2;
            continue;
        }
        if b[i..].starts_with(b"[[") {
            link += 1;
            i += 2;
            continue;
        }
        if b[i..].starts_with(b"]]") && link > 0 {
            link -= 1;
            i += 2;
            continue;
        }
        if b[i..].starts_with(b"{{{") {
            brace += 1;
            i += 3;
            continue;
        }
        if b[i..].starts_with(b"}}}") && brace > 0 {
            brace -= 1;
            i += 3;
            continue;
        }
        if b[i] == b'[' {
            bracket += 1;
        }
        if b[i] == b']' && bracket > 0 {
            bracket -= 1;
        }
        if link == 0 && brace == 0 && bracket == 0 && b[i..].starts_with(sep_b) {
            out.push(s[last..i].to_string());
            i += sep_b.len();
            last = i;
            continue;
        }
        i += 1;
    }
    out.push(s[last.min(s.len())..].to_string());
    out
}

impl Run {
    pub(crate) fn new(opt: Options, source: String) -> Run {
        Run { opt, source, footnotes: Vec::new(), named: HashMap::new(), categories: Vec::new(), headings: Vec::new(), section_counter: 0 }
    }

    fn run(mut self) -> Document {
        let lines = split_lines(&self.source);
        for line in &lines {
            let t = line.text.trim();
            if t.is_empty() {
                continue;
            }
            if let Some(c) = REDIRECT.captures(t) {
                if let Some(target) = ObjectRef::parse(&c[1]) {
                    return Document { source: self.source, blocks: Vec::new(), redirect: Some(target), categories: Vec::new(), sections: Vec::new(), footnotes: Vec::new() };
                }
            }
            break;
        }
        let mut blocks = self.blocks(&lines, 0, true);
        let sections = self.finish_headings(&mut blocks);
        Document { source: self.source, blocks, redirect: None, categories: self.categories, sections, footnotes: self.footnotes }
    }

    /// 문단 번호를 매기고 원문 범위를 정한다.
    fn finish_headings(&self, blocks: &mut [Block]) -> Vec<Section> {
        if self.headings.is_empty() {
            return Vec::new();
        }
        let min = self.headings.iter().map(|h| h.level).min().unwrap_or(1);
        let mut counters = [0usize; 7];
        let numbers: Vec<String> = self
            .headings
            .iter()
            .map(|h| {
                let d = (h.level - min) as usize;
                counters[d] += 1;
                for c in counters.iter_mut().skip(d + 1) {
                    *c = 0;
                }
                counters[..=d].iter().map(|n| n.to_string()).collect::<Vec<_>>().join(".")
            })
            .collect();
        let mut sections = Vec::new();
        for (i, h) in self.headings.iter().enumerate() {
            let end = self.headings[i + 1..].iter().find(|o| o.level <= h.level).map(|o| o.offset).unwrap_or(self.source.len());
            sections.push(Section { index: i, level: h.level, number: numbers[i].clone(), title: h.title.clone(), start: h.offset, end });
        }
        let mut k = 0;
        for b in blocks.iter_mut() {
            if let Block::Heading { number, section: Some(idx), .. } = b {
                *number = numbers[k].clone();
                *idx = k;
                k += 1;
            }
        }
        sections
    }

    pub(crate) fn blocks(&mut self, lines: &[Line], depth: usize, top: bool) -> Vec<Block> {
        let mut out = Vec::new();
        if depth > self.opt.max_depth {
            let text = lines.iter().map(|l| l.text.as_str()).collect::<Vec<_>>().join("\n");
            out.push(Block::Paragraph { children: vec![Inline::Text(text)], start: None, end: None });
            return out;
        }
        let mut i = 0;
        while i < lines.len() {
            let text = lines[i].text.as_str();
            let trimmed = text.trim();
            if trimmed.is_empty() || text.starts_with("##") {
                i += 1;
                continue;
            }
            if let Some((level, folded, title)) = heading(text) {
                let children = self.inline(title, depth + 1);
                let section = if top {
                    self.headings.push(HeadingInfo { level, offset: lines[i].offset, title: plain_inline(&children) });
                    self.section_counter += 1;
                    Some(self.section_counter - 1)
                } else {
                    None
                };
                out.push(Block::Heading { level, number: String::new(), section, folded, children });
                i += 1;
                continue;
            }
            if RULE.is_match(text) {
                out.push(Block::Rule);
                i += 1;
                continue;
            }
            if text.trim_start().starts_with("{{{") {
                let consumed = self.brace_block(lines, i, depth, &mut out);
                if consumed > 0 {
                    i += consumed;
                    continue;
                }
            }
            if is_table_start(text) {
                i += self.table(lines, i, depth, &mut out);
                continue;
            }
            if text.starts_with('>') {
                let mut quoted = Vec::new();
                while i < lines.len() && lines[i].text.starts_with('>') {
                    let (mut t, mut off) = (&lines[i].text[1..], lines[i].offset + 1);
                    if let Some(rest) = t.strip_prefix(' ') {
                        t = rest;
                        off += 1;
                    }
                    quoted.push(Line { text: t.to_string(), offset: off });
                    i += 1;
                }
                let inner = self.blocks(&quoted, depth + 1, false);
                out.push(Block::Quote(inner));
                continue;
            }
            if LIST.is_match(text) {
                i += self.list(lines, i, depth, &mut out);
                continue;
            }
            if text.starts_with(' ') {
                let mut group = Vec::new();
                let mut min_indent = usize::MAX;
                while i < lines.len() && lines[i].text.starts_with(' ') && !lines[i].text.trim().is_empty() && !LIST.is_match(&lines[i].text) {
                    let t = &lines[i].text;
                    min_indent = min_indent.min(t.len() - t.trim_start().len());
                    group.push(lines[i].clone());
                    i += 1;
                }
                let dedented: Vec<Line> = group
                    .iter()
                    .map(|l| {
                        let (t, cut) = dedent(&l.text, min_indent);
                        Line { text: t.to_string(), offset: l.offset + cut }
                    })
                    .collect();
                let inner = self.blocks(&dedented, depth + 1, false);
                out.push(Block::Indent { depth: min_indent, children: inner });
                continue;
            }
            if let Some(c) = BLOCK_MACRO.captures(trimmed) {
                let name = c[1].to_lowercase();
                if name == "목차" || name == "tableofcontents" {
                    out.push(Block::Macro("목차".into()));
                }
                if name == "각주" || name == "footnote" {
                    out.push(Block::Macro("각주".into()));
                }
                i += 1;
                continue;
            }
            if let Some(c) = INCLUDE.captures(trimmed) {
                let args = split_args(&c[1]);
                if let Some(target) = args.first().and_then(|a| ObjectRef::parse(a)) {
                    out.push(Block::Include { target, args: key_values(&args[1..]) });
                    i += 1;
                    continue;
                }
            }
            if let Some(c) = EMBED.captures(trimmed) {
                let args = split_args(&c[1]);
                if let Some(target) = args.first().and_then(|a| ObjectRef::parse(a)) {
                    out.push(Block::Embed { target, options: key_values(&args[1..]), raw: trimmed.to_string() });
                    i += 1;
                    continue;
                }
            }
            let first = i;
            let mut para: Vec<&str> = Vec::new();
            while i < lines.len() {
                let t = lines[i].text.as_str();
                if t.trim().is_empty() || t.starts_with("##") || self.starts_block(t) {
                    break;
                }
                para.push(t);
                i += 1;
            }
            if para.is_empty() {
                // 어떤 규칙에도 맞지 않는 시작 줄 — 글자로 남긴다.
                para.push(text);
                i += 1;
            }
            let joined = para.join("\n");
            let inlines = self.inline(&joined, depth + 1);
            if has_content(&inlines) {
                let end = lines[i - 1].offset + lines[i - 1].text.len();
                out.push(Block::Paragraph { children: inlines, start: Some(lines[first].offset), end: Some(end) });
            }
        }
        out
    }

    pub(crate) fn starts_block(&mut self, t: &str) -> bool {
        if heading(t).is_some() || RULE.is_match(t) {
            return true;
        }
        if is_table_start(t) || t.starts_with('>') || t.starts_with(' ') {
            return true;
        }
        let trimmed = t.trim();
        if BLOCK_MACRO.is_match(trimmed) || INCLUDE.is_match(trimmed) || EMBED.is_match(trimmed) {
            return true;
        }
        if t.starts_with("{{{") && self.match_braces(t.as_bytes(), t.find("{{{").unwrap_or(0)).is_none() {
            return true;
        }
        false
    }

    /// `start`의 `{{{`와 짝을 이루는 `}}}`의 시작 위치.
    pub(crate) fn match_braces(&mut self, b: &[u8], start: usize) -> Option<usize> {
        let (mut depth, mut i) = (0usize, start);
        while i < b.len() {
            if b[i..].starts_with(b"{{{") {
                depth += 1;
                i += 3;
                continue;
            }
            if b[i..].starts_with(b"}}}") {
                depth = depth.saturating_sub(1);
                if depth == 0 {
                    return Some(i);
                }
                i += 3;
                continue;
            }
            if b[i] == b'\\' {
                i += 2;
                continue;
            }
            i += 1;
        }
        None
    }

    /// 여러 줄에 걸친 `{{{ … }}}` 블록. 처리한 줄 수, 해당하지 않으면 0.
    fn brace_block(&mut self, lines: &[Line], i: usize, depth: usize, out: &mut Vec<Block>) -> usize {
        let first = lines[i].text.as_str();
        let lead = first.len() - first.trim_start().len();
        let is_bang = first[lead..].starts_with("{{{#!");
        // 한 줄 안에서 닫힌 `{{{ }}}`는 인라인이다. `#!` 블록만 예외.
        if !is_bang && self.match_braces(first.as_bytes(), lead).is_some() {
            return 0;
        }
        let mut all = String::new();
        let mut starts = Vec::new();
        for l in &lines[i..] {
            starts.push(all.len());
            all.push_str(&l.text);
            all.push('\n');
        }
        let Some(close) = self.match_braces(all.as_bytes(), lead) else {
            return 0;
        };
        let end_line = starts.iter().rposition(|&s| s <= close).unwrap_or(0);
        let header_end = all[lead..].find('\n').map(|p| p + lead);
        let (header, inner, inner_start) = match header_end {
            Some(he) if close >= he => {
                let inner_start = he + 1;
                let mut inner = if inner_start <= close { all[inner_start..close].to_string() } else { String::new() };
                if inner.ends_with('\n') {
                    inner.pop();
                }
                (all[lead + 3..he].to_string(), inner, inner_start)
            }
            _ => (all[lead + 3..close].to_string(), String::new(), close),
        };
        let mut inner_lines = Vec::new();
        let mut off = lines[i].offset + inner_start;
        for t in inner.split('\n') {
            inner_lines.push(Line { text: t.to_string(), offset: off });
            off += t.len() + 1;
        }
        let h = header.trim();
        if let Some(rest) = h.strip_prefix("#!folding") {
            let summary = rest.trim();
            let children = self.blocks(&inner_lines, depth + 1, false);
            out.push(Block::Fold { summary: if summary.is_empty() { "펼치기".into() } else { summary.to_string() }, children });
        } else if let Some(rest) = h.strip_prefix("#!syntax") {
            let lang = rest.trim();
            out.push(Block::Code { language: (!lang.is_empty()).then(|| lang.to_string()), text: inner.clone() });
        } else if h.starts_with("#!wiki") {
            let children = self.blocks(&inner_lines, depth + 1, false);
            out.push(Block::Indent { depth: 0, children });
        } else if h.starts_with("#!html") {
            out.push(Block::Unsupported { kind: "html".into(), text: inner.clone() });
        } else if h.starts_with("#!") {
            out.push(Block::Code { language: None, text: inner.clone() });
        } else if SIZE.is_match(&format!("{h} ")) || h.starts_with('#') {
            // 크기 · 색 블록. 색은 그리지 않는다.
            let children = self.blocks(&inner_lines, depth + 1, false);
            out.push(Block::Indent { depth: 0, children });
        } else {
            let text = if h.is_empty() { inner.clone() } else if inner.is_empty() { header.clone() } else { format!("{header}\n{inner}") };
            out.push(Block::Code { language: None, text });
        }
        let rest = all[close + 3..].split('\n').next().unwrap_or("");
        if !rest.trim().is_empty() {
            let para = self.inline(rest, depth + 1);
            if has_content(&para) {
                out.push(Block::Paragraph { children: para, start: None, end: None });
            }
        }
        end_line + 1
    }

    fn table(&mut self, lines: &[Line], start: usize, depth: usize, out: &mut Vec<Block>) -> usize {
        let mut i = start;
        let mut rows = Vec::new();
        let mut caption = None;
        while i < lines.len() {
            let t = lines[i].text.as_str();
            if !is_table_start(t) {
                break;
            }
            let mut buf = t.to_string();
            i += 1;
            while !row_closed(&buf) && i < lines.len() && !lines[i].text.starts_with("||") {
                buf.push('\n');
                buf.push_str(&lines[i].text);
                i += 1;
            }
            let mut row = buf.trim_end().to_string();
            if !row.starts_with("||") {
                if let Some(m) = CAPTION.captures(&row) {
                    let whole = m.get(0).map(|g| g.end()).unwrap_or(0);
                    if caption.is_none() {
                        let text = m[1].to_string();
                        caption = Some(self.inline(&text, depth + 1));
                    }
                    row = format!("|{}", &row[whole..]);
                    if !row.starts_with("||") {
                        row = format!("|{row}");
                    }
                }
            }
            // 옛 Dart는 끝 `||`를 먼저 떼고 앞 `||`를 뗐다 — `||` 하나뿐인 줄에서 범위를 넘었다. 앞부터 뗀다.
            let mut body = row.strip_prefix("||").unwrap_or(&row).to_string();
            if body.ends_with("||") {
                body.truncate(body.len() - 2);
            }
            rows.push(self.row(&body, depth));
        }
        out.push(Block::Table { caption, rows });
        i - start
    }

    fn row(&mut self, row: &str, depth: usize) -> Vec<Cell> {
        let parts = split_top_level(row, "||");
        let mut cells = Vec::new();
        let mut span = 0u32;
        for part in &parts {
            if part.is_empty() {
                span += 1;
                continue;
            }
            let mut content: &str = part;
            let (mut col_span, mut row_span, mut align) = (1 + span, 1u32, Align::Start);
            span = 0;
            loop {
                let trimmed = content.trim_start();
                if !trimmed.starts_with('<') {
                    break;
                }
                let Some(close) = trimmed.find('>') else { break };
                let attr = trimmed[1..close].trim();
                if let Some(c) = COLSPAN.captures(attr) {
                    col_span = c[1].parse::<u64>().map_or(64, |n| n.clamp(1, 64)) as u32;
                } else if let Some(c) = ROWSPAN.captures(attr) {
                    row_span = c[1].parse::<u64>().map_or(256, |n| n.clamp(1, 256)) as u32;
                } else if attr == ":" {
                    align = Align::Center;
                } else if attr == "(" {
                    align = Align::Start;
                } else if attr == ")" {
                    align = Align::End;
                } else if !CELL_ATTR.is_match(attr) {
                    break;
                }
                content = &trimmed[close + 1..];
            }
            let children = self.inline(content.trim(), depth + 1);
            cells.push(Cell { col_span, row_span, align, children });
        }
        cells
    }

    fn list(&mut self, lines: &[Line], start: usize, depth: usize, out: &mut Vec<Block>) -> usize {
        let mut entries: Vec<Entry> = Vec::new();
        let mut i = start;
        while i < lines.len() {
            let t = lines[i].text.as_str();
            if t.trim().is_empty() {
                break;
            }
            if let Some(c) = LIST.captures(t) {
                entries.push(Entry { indent: c[1].len(), marker: c[2].to_string(), start_at: c.get(3).and_then(|m| m.as_str().parse().ok()), text: c[4].to_string() });
            } else if t.starts_with(' ') && !entries.is_empty() {
                let e = entries.last_mut().unwrap();
                e.text.push('\n');
                e.text.push_str(t.trim());
            } else {
                break;
            }
            i += 1;
        }
        let mut pos = 0;
        while pos < entries.len() {
            let (block, next) = self.build_list(&entries, pos, entries[pos].indent, depth);
            out.push(block);
            pos = next;
        }
        i - start
    }

    pub(crate) fn build_list(&mut self, entries: &[Entry], start: usize, indent: usize, depth: usize) -> (Block, usize) {
        let marker = entries[start].marker.clone();
        let mut items: Vec<Vec<Block>> = Vec::new();
        let mut i = start;
        while i < entries.len() {
            let e = &entries[i];
            if e.indent < indent {
                break;
            }
            if e.indent > indent {
                if items.is_empty() {
                    items.push(Vec::new());
                }
                let (nested, next) = self.build_list(entries, i, e.indent, depth + 1);
                items.last_mut().unwrap().push(nested);
                i = next;
                continue;
            }
            if e.marker != marker && !items.is_empty() {
                break;
            }
            let inl = self.inline(&e.text, depth + 1);
            items.push(if has_content(&inl) { vec![Block::Paragraph { children: inl, start: None, end: None }] } else { Vec::new() });
            i += 1;
        }
        let style = match marker.as_str() {
            "1." => ListStyle::Decimal,
            "a." => ListStyle::LowerAlpha,
            "A." => ListStyle::UpperAlpha,
            "i." => ListStyle::LowerRoman,
            "I." => ListStyle::UpperRoman,
            _ => ListStyle::Bullet,
        };
        (Block::List { style, start: entries[start].start_at.unwrap_or(1), items }, i)
    }

    pub(crate) fn inline(&mut self, s: &str, depth: usize) -> Vec<Inline> {
        if depth > self.opt.max_depth {
            return vec![Inline::Text(s.to_string())];
        }
        let b = s.as_bytes();
        let mut out = Vec::new();
        let mut buf = String::new();
        let mut i = 0;
        'outer: while i < b.len() {
            let c = b[i];
            if c == b'\\' && i + 1 < b.len() {
                let ch = s[i + 1..].chars().next().unwrap();
                buf.push(ch);
                i += 1 + ch.len_utf8();
                continue;
            }
            if c == b'\n' {
                flush(&mut buf, &mut out);
                out.push(Inline::Break);
                i += 1;
                continue;
            }
            if b[i..].starts_with(b"'''''") {
                if let Some(rel) = s[i + 5..].find("'''''") {
                    let end = i + 5 + rel;
                    if end > i + 5 && !s[i + 5..end].contains('\n') {
                        flush(&mut buf, &mut out);
                        let inner = self.inline(&s[i + 5..end], depth + 1);
                        out.push(Inline::Styled { style: Style::Bold, children: vec![Inline::Styled { style: Style::Italic, children: inner }] });
                        i = end + 5;
                        continue;
                    }
                }
            }
            for (marker, style) in PAIRED {
                if !b[i..].starts_with(marker.as_bytes()) {
                    continue;
                }
                let from = i + marker.len();
                if let Some(rel) = s[from..].find(marker) {
                    let end = from + rel;
                    if end > from && !s[from..end].contains('\n') {
                        flush(&mut buf, &mut out);
                        let inner = self.inline(&s[from..end], depth + 1);
                        out.push(Inline::Styled { style, children: inner });
                        i = end + marker.len();
                        continue 'outer;
                    }
                }
                break;
            }
            if b[i..].starts_with(b"{{{") {
                if let Some(end) = self.match_braces(b, i) {
                    flush(&mut buf, &mut out);
                    let inner = &s[i + 3..end];
                    if let Some(m) = SIZE.captures(inner) {
                        let step = m[2].parse::<i8>().unwrap_or(1) * if &m[1] == "+" { 1 } else { -1 };
                        let rest = &inner[m.get(0).map(|g| g.end()).unwrap_or(0)..];
                        let children = self.inline(rest, depth + 1);
                        out.push(Inline::Size { step, children });
                    } else if inner.starts_with('#') && !inner.starts_with("#!") {
                        let content = inner.find(' ').map(|sp| &inner[sp + 1..]).unwrap_or("");
                        let children = self.inline(content, depth + 1);
                        out.extend(children);
                    } else {
                        out.push(Inline::Code(inner.to_string()));
                    }
                    i = end + 3;
                    continue;
                }
            }
            if b[i..].starts_with(b"[[") {
                if let Some(end) = self.match_pair(b, i) {
                    flush(&mut buf, &mut out);
                    if let Some(node) = self.link(&s[i + 2..end], depth) {
                        out.push(node);
                    }
                    i = end + 2;
                    continue;
                }
            }
            if b[i..].starts_with(b"[*") {
                if let Some(end) = self.match_bracket(b, i) {
                    flush(&mut buf, &mut out);
                    let node = self.footnote(&s[i + 2..end], depth);
                    out.push(node);
                    i = end + 1;
                    continue;
                }
            }
            if c == b'[' {
                if let Some(m) = MACRO.captures(&s[i..]) {
                    if let Some(node) = macro_node(&m[1], m.get(2).map(|g| g.as_str())) {
                        flush(&mut buf, &mut out);
                        out.push(node);
                        i += m.get(0).map(|g| g.end()).unwrap_or(1);
                        continue;
                    }
                }
            }
            if self.opt.template && c == b'@' {
                if let Some(m) = PARAM.captures(&s[i..]) {
                    flush(&mut buf, &mut out);
                    out.push(Inline::Param { name: m[1].to_string(), fallback: m.get(2).map(|g| g.as_str().to_string()) });
                    i += m.get(0).map(|g| g.end()).unwrap_or(1);
                    continue;
                }
            }
            let ch = s[i..].chars().next().unwrap();
            buf.push(ch);
            i += ch.len_utf8();
        }
        flush(&mut buf, &mut out);
        out
    }

    /// `[[`와 짝인 `]]`의 시작 위치. 바깥 단계에서 줄이 바뀌면 없음.
    pub(crate) fn match_pair(&mut self, b: &[u8], start: usize) -> Option<usize> {
        let (mut depth, mut i) = (0usize, start);
        while i < b.len() {
            if b[i] == b'\\' {
                i += 2;
                continue;
            }
            if b[i..].starts_with(b"[[") {
                depth += 1;
                i += 2;
                continue;
            }
            if b[i..].starts_with(b"]]") {
                depth = depth.saturating_sub(1);
                if depth == 0 {
                    return Some(i);
                }
                i += 2;
                continue;
            }
            if b[i] == b'\n' && depth <= 1 {
                return None;
            }
            i += 1;
        }
        None
    }

    /// `[*`의 `[`와 짝인 `]`의 위치.
    pub(crate) fn match_bracket(&mut self, b: &[u8], start: usize) -> Option<usize> {
        let (mut depth, mut i) = (0usize, start);
        while i < b.len() {
            match b[i] {
                b'\\' => {
                    i += 2;
                    continue;
                }
                b'[' => depth += 1,
                b']' => {
                    depth = depth.saturating_sub(1);
                    if depth == 0 {
                        return Some(i);
                    }
                }
                _ => {}
            }
            i += 1;
        }
        None
    }

    fn link(&mut self, inner: &str, depth: usize) -> Option<Inline> {
        let parts = split_top_level(inner, "|");
        let target_text = parts[0].trim();
        let label = (parts.len() > 1).then(|| parts[1..].join("|"));
        if target_text.starts_with("http://") || target_text.starts_with("https://") {
            let label = label.map(|l| self.inline(&l, depth + 1));
            return Some(Inline::Url { uri: target_text.to_string(), label });
        }
        if let Some(anchor) = target_text.strip_prefix('#') {
            let anchor = anchor.trim();
            let label = label.map(|l| self.inline(&l, depth + 1));
            let target = ObjectRef { ns: String::new(), key: String::new(), anchor: (!anchor.is_empty()).then(|| anchor.to_string()) };
            return Some(Inline::Link { target, label });
        }
        let mut target = target_text.to_string();
        let escaped = target.starts_with(':');
        if escaped {
            target.remove(0);
        }
        if let Some(current) = self.opt.current.clone() {
            if target.starts_with('/') {
                target = format!("{current}{target}");
            } else if let Some(rest) = target.strip_prefix("..") {
                if rest.starts_with('/') {
                    let parent = current.rfind('/').map(|cut| &current[..cut]).unwrap_or(&current);
                    target = format!("{parent}{rest}");
                    if target.ends_with('/') {
                        target.pop();
                    }
                }
            }
        }
        let Some(r) = ObjectRef::parse(&target) else {
            return Some(Inline::Text(format!("[[{inner}]]")));
        };
        if !escaped && r.ns == CATEGORY {
            if !self.categories.contains(&r.key) {
                self.categories.push(r.key);
            }
            return None;
        }
        if !escaped && r.ns == FILE {
            return Some(Inline::File { file: r.without_anchor(), options: options(label.as_deref()) });
        }
        let label = label.map(|l| self.inline(&l, depth + 1));
        Some(Inline::Link { target: r, label })
    }

    fn footnote(&mut self, inner: &str, depth: usize) -> Inline {
        let (name, content): (Option<String>, &str) = if !inner.is_empty() && !inner.starts_with(' ') {
            match inner.find(' ') {
                Some(sp) => (Some(inner[..sp].to_string()), &inner[sp + 1..]),
                None => (Some(inner.to_string()), ""),
            }
        } else {
            (None, inner.strip_prefix(' ').unwrap_or(inner))
        };
        let reuse = name.as_ref().filter(|_| content.trim().is_empty()).and_then(|n| self.named.get(n).copied());
        if let Some(number) = reuse {
            return Inline::FootnoteRef { number, name };
        }
        // 번호를 먼저 잡아 둔다 — 옛 Dart는 안쪽 각주가 같은 번호를 받았다.
        let number = self.footnotes.len() as u32 + 1;
        self.footnotes.push(FootnoteDef { number, name: name.clone(), children: Vec::new() });
        if let Some(n) = &name {
            self.named.entry(n.clone()).or_insert(number);
        }
        let children = self.inline(content, depth + 1);
        self.footnotes[(number - 1) as usize].children = children;
        Inline::FootnoteRef { number, name }
    }
}

fn row_closed(row: &str) -> bool {
    let r = row.trim_end();
    r.len() >= 4 && r.ends_with("||")
}

fn macro_node(name: &str, args: Option<&str>) -> Option<Inline> {
    let lower = name.to_lowercase();
    match lower.as_str() {
        "br" => Some(Inline::Break),
        "date" | "datetime" => Some(Inline::Macro { name: "date".into(), args: Vec::new() }),
        "age" | "dday" | "anchor" => args.map(|a| Inline::Macro { name: lower.clone(), args: vec![a.trim().to_string()] }),
        "ruby" => {
            let parts = split_args(args?);
            let base = parts.first()?.clone();
            let kv = key_values(&parts[1..]);
            Some(Inline::Ruby { base, ruby: kv.get("ruby").cloned().unwrap_or_default() })
        }
        _ => None,
    }
}

    pub(crate) fn match_braces(&mut self, b: &[u8], start: usize) -> Option<usize> {

        let (mut depth, mut i) = (0usize, start);

fn plain_blocks(blocks: &[Block], out: &mut String) {
    for b in blocks {
        match b {
            Block::Heading { children, .. } | Block::Paragraph { children, .. } => {
                out.push_str(&plain_inline(children));
                out.push('\n');
            }
            Block::List { items, .. } => items.iter().for_each(|item| plain_blocks(item, out)),
            Block::Quote(children) | Block::Indent { children, .. } => plain_blocks(children, out),
            Block::Table { caption, rows } => {
                if let Some(c) = caption {
                    out.push_str(&plain_inline(c));
                    out.push('\n');
                }
                for row in rows {
                    out.push_str(&row.iter().map(|c| plain_inline(&c.children)).collect::<Vec<_>>().join(" "));
                    out.push('\n');
                }
            }
            Block::Code { text, .. } | Block::Unsupported { text, .. } => {
                out.push_str(text);
                out.push('\n');
            }
            Block::Fold { summary, children } => {
                out.push_str(summary);
                out.push('\n');
                plain_blocks(children, out);
            }

/// 옛 Dart와 같은 차례 — 먼저 `[ \t]+` → 공백 하나, 그다음 `\n{2,}` → `\n`, 앞뒤를 걷는다.
fn collapse(s: &str) -> String {
    let mut spaced = String::with_capacity(s.len());
    let mut in_space = false;
    for c in s.chars() {
        if c == ' ' || c == '\t' {
            if !in_space {
                spaced.push(' ');
            }
            in_space = true;
        } else {
            spaced.push(c);
            in_space = false;
        }
    }
    let mut out = String::with_capacity(spaced.len());
    let mut newlines = 0;
    for c in spaced.chars() {
        if c == '\n' {
            newlines += 1;
            if newlines == 1 {
                out.push('\n');
            }
        } else {
            newlines = 0;
            out.push(c);
        }
    }
    out.trim().to_string()
}

/// 이 문서가 가리키는 다른 대상. 같은 대상은 종류마다 한 번, 앵커는 뺀다. 같은 문서 안 링크는 넣지 않는다.
pub fn out_links(doc: &Document) -> Vec<OutLink> {
    let mut w = LinkWalk { seen: HashSet::new(), out: Vec::new() };
    if let Some(r) = &doc.redirect {
        w.add(r, LinkKind::Redirect);
    }
    w.blocks(&doc.blocks);
    for f in &doc.footnotes {
        w.inlines(&f.children);
    }
    w.out
}

struct LinkWalk {
    seen: HashSet<(LinkKind, String)>,
    out: Vec<OutLink>,
}

impl LinkWalk {
    fn add(&mut self, r: &ObjectRef, kind: LinkKind) {
        if r.key.is_empty() {
            return;
        }
        let r = r.without_anchor();
        if self.seen.insert((kind, r.path())) {
            self.out.push(OutLink { target: r, kind });
        }
    }

    fn inlines(&mut self, nodes: &[Inline]) {
        for n in nodes {
            match n {
                Inline::Link { target, label } => {
                    if !target.is_same_page() {
                        self.add(target, LinkKind::Body);
                    }
                    if let Some(l) = label {
                        self.inlines(l);
                    }
                }
                Inline::File { file, .. } => self.add(file, LinkKind::File),
                Inline::Styled { children, .. } | Inline::Size { children, .. } => self.inlines(children),
                Inline::Url { label: Some(l), .. } => self.inlines(l),
                _ => {}
            }
        }
    }

    fn blocks(&mut self, blocks: &[Block]) {
        for b in blocks {
            match b {
                Block::Heading { children, .. } | Block::Paragraph { children, .. } => self.inlines(children),
                Block::List { items, .. } => items.iter().for_each(|item| self.blocks(item)),
                Block::Quote(children) | Block::Indent { children, .. } | Block::Fold { children, .. } => self.blocks(children),
                Block::Table { caption, rows } => {
                    if let Some(c) = caption {
                        self.inlines(c);
                    }
                    for row in rows {
                        for cell in row {
                            self.inlines(&cell.children);
                        }
                    }
                }
                Block::Include { target, .. } => self.add(target, LinkKind::Include),
                Block::Embed { target, .. } => self.add(target, LinkKind::Embed),
                _ => {}
            }
        }
    }
}

pub fn timeline_entities(doc: &Document) -> Vec<String> {
    fn inlines(nodes: &[Inline], out: &mut Vec<String>) {
        for n in nodes {
            match n {
                Inline::Link { target, label } => {
                    if target.ns == TIMELINE && !target.key.is_empty() && !out.contains(&target.key) {
                        out.push(target.key.clone());
                    }
                    if let Some(l) = label {
                        inlines(l, out);
                    }
                }
                Inline::Styled { children, .. } | Inline::Size { children, .. } => inlines(children, out),
                Inline::Url { label: Some(l), .. } => inlines(l, out),
                _ => {}
            }
        }
    }
    fn blocks(list: &[Block], out: &mut Vec<String>) {
        for b in list {
            match b {
                Block::Heading { children, .. } | Block::Paragraph { children, .. } => inlines(children, out),
                Block::List { items, .. } => items.iter().for_each(|i| blocks(i, out)),
                Block::Quote(c) | Block::Indent { children: c, .. } | Block::Fold { children: c, .. } => blocks(c, out),
                Block::Table { caption, rows } => {
                    if let Some(c) = caption {
                        inlines(c, out);
                    }
                    rows.iter().flatten().for_each(|cell| inlines(&cell.children, out));
                }
                Block::TimelineStrip { entity, .. } => {
                    if !out.contains(entity) {
                        out.push(entity.clone());
                    }
                }
                _ => {}
            }
        }
    }
    let mut out = Vec::new();
    blocks(&doc.blocks, &mut out);
    for f in &doc.footnotes {
        inlines(&f.children, &mut out);

static FILL: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"@([^@\s=]+)(?:=([^@\n]*))?@").unwrap());

/// 틀 원문의 `@이름@` · `@이름=기본값@`을 해석하기 전에 글로 채운다(나무위키처럼 — 인자 안의 표식도 해석된다).
fn fill_params(source: &str, args: &BTreeMap<String, String>) -> String {
    FILL.replace_all(source, |c: &regex::Captures| match args.get(&c[1]) {
        Some(v) => v.clone(),
        None => c.get(2).map(|g| g.as_str().to_string()).unwrap_or_default(),
    })
    .into_owned()
}

fn is_han(c: char) -> bool {
    matches!(c as u32, 0x4E00..=0x9FFF | 0x3400..=0x4DBF | 0xF900..=0xFAFF | 0x20000..=0x323AF)
}

/// 글을 한자 구간과 나머지로 가른다.
fn split_han(t: &str, out: &mut Vec<MInline>) {
    let mut run = String::new();
    let mut han = false;
    for c in t.chars() {
        let h = is_han(c);
        if h != han && !run.is_empty() {
            let s = std::mem::take(&mut run);
            out.push(if han { MInline::Han { s } } else { MInline::Text { s } });
        }
        han = h;
        run.push(c);
    }
    if !run.is_empty() {
        out.push(if han { MInline::Han { s: run } } else { MInline::Text { s: run } });
    }
}

/// 바이트 위치들을 UTF-16 칸 위치로 — 한 번 훑는다.
fn utf16_positions(src: &str, bytes: &[usize]) -> Vec<usize> {
    let mut want: Vec<(usize, usize)> = bytes.iter().copied().enumerate().map(|(i, b)| (b, i)).collect();
    want.sort();
    let mut out = vec![0; bytes.len()];
    let (mut units, mut k) = (0usize, 0usize);
    for (at, c) in src.char_indices() {
        while k < want.len() && want[k].0 <= at {
            out[want[k].1] = units;
            k += 1;
        }
        units += c.len_utf16();
    }
    while k < want.len() {
        out[want[k].1] = units;
        k += 1;
    }
    out
}

/// 원문을 그리기 모델로.
pub fn render(source: &str, options: &RenderOptions, templates: &dyn Templates) -> Model {
    let doc = parse(source, &Options { template: options.template, current: options.title.clone(), ..Options::default() });
    if let Some(r) = doc.redirect {
        return Model { version: MODEL_VERSION, redirect: Some(r), blocks: Vec::new(), footnotes: Vec::new(), sections: Vec::new(), categories: Vec::new(), template_categories: Vec::new(), templates: Vec::new() };
    }
    let mut chain = Vec::new();
    if let (true, Some(t)) = (options.template, &options.title) {
        chain.push(format!("{TEMPLATE}:{t}"));
    }
    let mut b = Builder { templates, chain, expansions: 0, defs: vec![doc.footnotes.clone()], numbers: HashMap::new(), footnotes: Vec::new(), template_categories: Vec::new(), used: Vec::new() };
    let blocks = b.blocks(&doc.blocks, 0);
    let mut bytes = Vec::with_capacity(doc.sections.len() * 2);
    for s in &doc.sections {
        bytes.push(s.start);
        bytes.push(s.end);
    }
    let units = utf16_positions(&doc.source, &bytes);
    let sections = doc
        .sections
        .iter()
        .enumerate()
        .map(|(i, s)| MSection { index: s.index, level: s.level, number: s.number.clone(), title: s.title.clone(), start: units[2 * i], end: units[2 * i + 1] })
        .collect();
    Model { version: MODEL_VERSION, redirect: None, blocks, footnotes: b.footnotes, sections, categories: doc.categories, template_categories: b.template_categories, templates: b.used }
}

struct Builder<'t> {
    templates: &'t dyn Templates,
    /// 지금 펼치는 중인 틀의 사슬(고리 막기).
    chain: Vec<String>,
    expansions: usize,
    /// 원문마다(0 = 문서, 1.. = 펼친 틀 하나하나)의 각주 정의.
    defs: Vec<Vec<FootnoteDef>>,
    /// (원문, 원문 속 번호) → 모델 번호.
    numbers: HashMap<(usize, u32), u32>,
    footnotes: Vec<MFootnote>,
    template_categories: Vec<String>,
    used: Vec<String>,
}

impl Builder<'_> {
    fn blocks(&mut self, blocks: &[Block], inst: usize) -> Vec<MBlock> {
        let mut out = Vec::with_capacity(blocks.len());
        for b in blocks {
            let m = match b {
                Block::Heading { level, number, section, folded, children } => MBlock::Heading {
                    level: *level,
                    number: if inst == 0 { number.clone() } else { String::new() },
                    section: if inst == 0 { *section } else { None },
                    folded: *folded,
                    inl: self.inlines(children, inst),
                },
                Block::Paragraph { children, .. } => MBlock::Para { inl: self.inlines(children, inst) },
                Block::List { style, start, items } => MBlock::List {
                    style: match style {
                        ListStyle::Bullet => MListStyle::Bullet,
                        ListStyle::Decimal => MListStyle::Decimal,
                        ListStyle::LowerAlpha => MListStyle::LowerAlpha,
                        ListStyle::UpperAlpha => MListStyle::UpperAlpha,
                        ListStyle::LowerRoman => MListStyle::LowerRoman,
                        ListStyle::UpperRoman => MListStyle::UpperRoman,
                    },
                    start: *start,
                    items: items.iter().map(|item| self.blocks(item, inst)).collect(),
                },
                Block::Quote(children) => MBlock::Quote { blocks: self.blocks(children, inst) },
                Block::Table { caption, rows } => MBlock::Table {
                    caption: caption.as_ref().map(|c| self.inlines(c, inst)),
                    rows: rows
                        .iter()
                        .map(|row| {
                            row.iter()
                                .map(|cell| MCell {
                                    col: cell.col_span,
                                    row: cell.row_span,
                                    align: match cell.align {
                                        Align::Start => MAlign::Start,
                                        Align::Center => MAlign::Center,
                                        Align::End => MAlign::End,
                                    },
                                    inl: self.inlines(&cell.children, inst),
                                })
                                .collect()
                        })
                        .collect(),
                },
                Block::Code { language, text } => MBlock::Code { lang: language.clone(), text: text.clone() },
                Block::Fold { summary, children } => MBlock::Fold { summary: summary.clone(), blocks: self.blocks(children, inst) },
                Block::Indent { depth, children } => MBlock::Indent { depth: *depth, blocks: self.blocks(children, inst) },
                Block::Rule => MBlock::Rule,
                Block::Include { target, args } => self.include(target, args),
                Block::Embed { target, options, .. } => MBlock::Embed { target: target.clone(), options: options.clone() },
                Block::Macro(name) if name == "목차" => MBlock::Toc,
                Block::Macro(_) => MBlock::Footnotes,
                Block::Front { name, arg } => MBlock::Front { name: name.clone(), arg: arg.clone() },
                Block::TimelineStrip { entity, range } => MBlock::TimelineStrip { entity: entity.clone(), range: range.clone() },
                Block::Unsupported { kind, text } => MBlock::Unsupported { kind: kind.clone(), text: text.clone() },
            };
            out.push(m);
        }
        out
    }

    fn include(&mut self, target: &ObjectRef, args: &BTreeMap<String, String>) -> MBlock {
        let path = target.path();
        self.note(&path);
        let stop = |status| MBlock::Template { target: target.clone(), status, blocks: Vec::new() };
        if self.chain.contains(&path) {
            return stop(TemplateStatus::Cycle);
        }
        if self.chain.len() >= MAX_TEMPLATE_DEPTH {
            return stop(TemplateStatus::TooDeep);
        }
        if self.expansions >= MAX_TEMPLATE_EXPANSIONS {
            return stop(TemplateStatus::TooMany);
        }
        let Some(mut source) = self.templates.source(target) else {
            return stop(TemplateStatus::Missing);
        };
        let mut real = target.clone();
        if let Some(next) = redirect_of(&source) {
            // 이름 바뀐 틀 — 넘겨주기는 한 번만 따라간다.
            let next_path = next.path();
            self.note(&next_path);
            if next_path == path || self.chain.contains(&next_path) {
                return stop(TemplateStatus::Cycle);
            }
            match self.templates.source(&next) {
                Some(s) if redirect_of(&s).is_none() => {
                    source = s;
                    real = next;
                }
                _ => return stop(TemplateStatus::Missing),
            }
        }
        self.expansions += 1;
        let filled = fill_params(&source, args);
        let tdoc = parse(&filled, &Options { template: false, current: Some(real.key.clone()), ..Options::default() });
        for c in &tdoc.categories {
            if !self.template_categories.contains(c) {
                self.template_categories.push(c.clone());
            }
        }
        let inst = self.defs.len();
        self.defs.push(tdoc.footnotes.clone());
        self.chain.push(real.path());
        let blocks = self.blocks(&tdoc.blocks, inst);
        self.chain.pop();
        MBlock::Template { target: target.clone(), status: TemplateStatus::Ok, blocks }
    }

    fn note(&mut self, path: &str) {
        if !self.used.iter().any(|u| u == path) {
            self.used.push(path.to_string());
        }
    }

    fn inlines(&mut self, nodes: &[Inline], inst: usize) -> Vec<MInline> {
        let mut out = Vec::with_capacity(nodes.len());
        for n in nodes {
            match n {
                Inline::Text(t) => split_han(t, &mut out),
                Inline::Styled { style, children } => {
                    let style = match style {
                        Style::Bold => MStyle::Bold,
                        Style::Italic => MStyle::Italic,
                        Style::Underline => MStyle::Underline,
                        Style::Strike => MStyle::Strike,
                        Style::Superscript => MStyle::Sup,
                        Style::Subscript => MStyle::Sub,
                    };
                    let inl = self.inlines(children, inst);
                    out.push(MInline::Style { style, inl });
                }
                Inline::Size { step, children } => {
                    let inl = self.inlines(children, inst);
                    out.push(MInline::Size { step: *step, inl });
                }
                Inline::Link { target, label } => {
                    let inl = label.as_ref().map(|l| self.inlines(l, inst));
                    out.push(MInline::Link { target: target.clone(), inl });
                }
                Inline::Url { uri, label } => {
                    let inl = label.as_ref().map(|l| self.inlines(l, inst));
                    out.push(MInline::Url { href: uri.clone(), inl });
                }
                Inline::File { file, options } => out.push(MInline::File { key: file.key.clone(), options: options.clone() }),
                Inline::FootnoteRef { number, .. } => {
                    let n = self.footnote(inst, *number);
                    out.push(MInline::Fn { n });
                }
                Inline::Ruby { base, ruby } => out.push(MInline::Ruby { base: base.clone(), ruby: ruby.clone() }),
                Inline::Code(t) => out.push(MInline::Code { s: t.clone() }),
                Inline::Break => out.push(MInline::Br),
                Inline::Macro { name, args } => out.push(MInline::Macro { name: name.clone(), args: args.clone() }),
                Inline::Param { name, fallback } => out.push(MInline::Param { name: name.clone(), fallback: fallback.clone() }),
            }
        }
        out
    }

    /// 그린 차례로 각주 번호를 매긴다. 같은 원문의 같은 번호(이름 붙인 각주의 재사용)는 같은 번호.
    fn footnote(&mut self, inst: usize, number: u32) -> u32 {
        if let Some(&n) = self.numbers.get(&(inst, number)) {
            return n;
        }
        let n = self.footnotes.len() as u32 + 1;
        self.numbers.insert((inst, number), n);
        let def = self.defs[inst].iter().find(|d| d.number == number).cloned();
        self.footnotes.push(MFootnote { n, name: def.as_ref().and_then(|d| d.name.clone()), inl: Vec::new() });
        let inl = match &def {
            Some(d) => self.inlines(&d.children, inst),
            None => Vec::new(),
        };
        self.footnotes[(n - 1) as usize].inl = inl;
        n
    }
}

impl Templates for LibraryTemplates<'_> {
    fn source(&self, target: &ObjectRef) -> Option<String> {
        let ns = Namespace::ALL.into_iter().find(|n| n.prefix() == target.ns)?;

/// 저장하지 않은 글의 그리기 모델(편집기 미리보기 · spec/namumark.md). `handle`이 있으면 그 서재의 틀을 펼친다.
fn parse_op(req: &Value) -> Answer {
    let text = text(req, "text")?.to_string();
    let options = RenderOptions {
        title: req.get("title").and_then(Value::as_str).map(str::to_string),
        template: req.get("ns").and_then(Value::as_str) == Some(Namespace::Template.code()),

    } else {
        json!(render(&text, &options, &NoTemplates))
    };
    Ok(json!({"ok": true, "model": model}))
}

## 틀 펼치기

- `[include(틀:이름, 인자=값)]`는 저장할 때 펼쳐 모델에 굳힌다. 틀 원문의 `@이름@` · `@이름=기본값@`을 **해석하기 전에** 인자 글로 채운다(인자가 없고 기본값도 없으면 빈 글).
- 틀이 넘겨주기면 **한 번** 따라간다(이름 바꾼 틀). 두 번은 따르지 않는다(`missing`).
- 멈춤: 펼치는 사슬에 이미 있는 틀(`cycle`) · 깊이 8(`too_deep`) · 한 문서에서 2,000번(`too_many`) · 없는 틀(`missing`). 그 자리는 `blocks`가 빈 `template` 블록.
- 틀의 각주는 문서의 각주와 함께 **그린 차례로** 번호를 다시 매긴다. 틀의 분류는 `template_categories`(집은 본문 첫 분류).
- `templates`는 펼치려 한 틀 전부(없던 것 · 넘겨주기 대상 포함) — 그 틀이 생기거나 바뀌면 이 문서를 다시 그린다(설계 5-5).
